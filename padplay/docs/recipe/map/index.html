using PadPlay;
using UnityEngine;
using UnityEngine.EventSystems;

// On a map UI element: a tap on the phone is an ordinary uGUI click.
public class MapPin : MonoBehaviour, IPointerClickHandler
{
    [SerializeField] PadPlayHost host;

    public void OnPointerClick(PointerEventData e)
    {
        if (e is PadPlayPointerEventData p)
            host.Pulse(p.PlayerSlot, 80);   // short buzz for whoever tapped
    }
}
