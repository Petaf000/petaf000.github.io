PadPlay

2 人以上で遊ぶ

スマホが参加したらキャラクターを出し、本当に抜けたら消します。

Host と同じ GameObject に、下のスクリプト Players を追加する

using System.Collections.Generic;
using PadPlay;
using UnityEngine;

public class Players : MonoBehaviour
{
    [SerializeField] PadPlayHost host;
    [SerializeField] GameObject playerPrefab;
    readonly Dictionary<int, GameObject> _bySlot = new Dictionary<int, GameObject>();

    void OnEnable()
    {
        host.PlayerReady += OnReady;
        host.PlayerRemoved += OnRemoved;
    }

    void OnDisable()
    {
        host.PlayerReady -= OnReady;
        host.PlayerRemoved -= OnRemoved;
    }

    void OnReady(PadPlayPlayer p)
    {
        if (_bySlot.ContainsKey(p.PlayerSlot))
            return;   // came back: keep the same character
        _bySlot[p.PlayerSlot] = Instantiate(playerPrefab);
    }

    void OnRemoved(PadPlayPlayerRemoved r)
    {
        if (_bySlot.Remove(r.PlayerSlot, out GameObject character))
            Destroy(character);
    }
}

2 台目のスマホの Wi-Fi を切っても、キャラクターは残ります。300 秒たつか、アプリで「切る」を押すと消えます。

  • host.PlayerReady
  • host.PlayerRemoved
  • leftremoved
  • host.Touch.SetPlayerTarget
  • 2 台目のスマホが入れない
